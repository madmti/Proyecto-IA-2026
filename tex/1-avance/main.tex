\begin{frame}{Problema, representación y restricciones}
  \textbf{Problema:} Dado un histograma de 256 niveles y un $k$ fijo, elegir $k$
  umbrales que maximicen la varianza entre clases según el criterio de
  Otsu~\cite{Otsu1979}.

  \textbf{Representación de una solución:}
  $t = (t_1, t_2, \ldots, t_k)$ un vector de $k$ umbrales

  \textbf{Dominios:} $\forall\, t_i \in \{1, \ldots, 254\} \subset \mathbb{Z}$

  \textbf{Restricciones:}
  \begin{itemize}
    \item Cota inferior: $1 \leq t_1$
    \item Monotonía estricta: $t_i < t_{i+1}\ \forall\, i = 1, \ldots, k-1$
    \item Cota superior: $t_k \leq 254$
  \end{itemize}

\end{frame}

\begin{frame}{Problema, representación y restricciones}
  \textbf{Implementación:}
  \begin{itemize}
    \item Instancia:
    \begin{itemize}
      \item \texttt{std::array<unsigned long long, 256> histograma}
      \item \texttt{int k}
    \end{itemize}
    \item Solución: \texttt{std::vector<int> umbrales}
    \begin{itemize}
      \item No garantiza orden en los umbrales (\texttt{std::set}) pero es mejor en la reparación
      \item No tiene tamaño fijo a pesar de un $k$ fijo (\texttt{std::array}) pero es mejor en la generación
      \item Garantiza umbrales enteros
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Problema, representación y restricciones}
  \textbf{Manejo de restricciones --- Generación de solución inicial:}

  Greedy busca secuencialmente los umbrales, se inserta uno artificial 0 al inicio del vector
\begin{minted}[highlightlines={2,5}]{cpp}

\end{minted}
\end{frame}

\begin{frame}[fragile]{Problema, representación y restricciones}
  \textbf{Manejo de restricciones --- Generación de solución inicial:}

  En cada iteración se setea la cota inferior con respecto al umbral anterior que conserva la monotonía y una cota superior hasta el máximo posible para cada umbral.
\begin{minted}[highlightlines={4,6}]{cpp}
void greedy(std::vector<int> &umbrales, double &facc, int k, const otsu::TablaOtsu &tabla, std::mt19937 &gen) {
  int size = static_cast<int>(umbrales.size());
  ..
  int cotainf = umbrales.back() + 1, cota_sup = 254 - k + size;
  ..
  for (int um = cotainf; um <= cota_sup; um++) {
    ..
  };
  ..
  greedy(umbrales, facc, k, tabla, gen);
}
\end{minted}
\end{frame}

\begin{frame}[fragile]{Problema, representación y restricciones}
  \textbf{Manejo de restricciones --- Reparación de la solución:}

  Tabú search filtra los valores posibles para los vecinos entre la cota superior e inferior. Asegura monotonía estricta usando pasos de $\pm 1$ y filtrando vecinos que ya existan en el vector de umbrales ($t_i + \text{mov} \notin t$).
\begin{minted}[fontsize=\scriptsize, highlightlines={7-10,15-16}]{cpp}
std::vector<std::tuple<int, int>> generarVecinos(std::vector<int> &actual, int size) {
  std::vector<std::tuple<int, int>> vecinos;

  auto agregarVecinos = [&actual, &vecinos, &size](int mov) {
    for (int i = 0; i < size; i++) {
      int val = actual[i] + mov;
      if (val < 1 || val > 254)
        continue;
      if (std::find(actual.begin(), actual.end(), val) != actual.end())
        continue;
      ..
    }
  };

  agregarVecinos(1);
  agregarVecinos(-1);
  return vecinos;
}
\end{minted}
\end{frame}

\begin{frame}{Función de evaluación}
  \textbf{Objetivo:} $f(t) = \sum_{j=0}^{k} \omega_j (\mu_j - \mu_T)^2$ con clases $C_j = [t_j, t_{j+1})$, $t_0 = 0$, $t_{k+1} = 256$

  \textbf{Problema:} evaluar directo cuesta $O(256)$ por solución, y la búsqueda evalúa miles.

  \textbf{Solución:} tablas de sumas acumuladas \texttt{acumP} y \texttt{acumGP}~\cite{Liao2001Fast}
  \begin{itemize}
    \item Se calculan una vez: $O(256)$
    \item Cada clase es una resta de dos prefijos: evaluar cuesta $O(k)$
    \item Clase vacía ($\omega_j = 0$): aporta $0$
    \item Verificada contra una implementación directa
  \end{itemize}

\begin{frame}{Solución inicial: Greedy}
  \begin{itemize}
    \item Construye la solución de forma secuencial, partiendo con \texttt{umbrales = \{0\}} (umbral artificial)
    \item Dado el umbral anterior $t_{i-1}$, prueba cada candidato $t_i'$ con
    \[ t_i' \in [\,t_{i-1}+1,\ 254-k+i\,] \]
    (la cota superior deja espacio a los $k-i$ umbrales restantes)
    \item Aporte de cada candidato:
    \[ a(t_i') = f_T(t_{i-1}, t_i') + f_T(t_i', 256) \]
    clase que se cierra $+$ toda la masa restante tratada como una sola clase abierta
    \item Elige el mayor aporte; en caso de empate, al azar con una semilla
    \item Termina con $k$ umbrales (más el 0, que se retira)
  \end{itemize}
  \textbf{Coherencia:} usa el mismo $f_T$ y las mismas tablas de la evaluación ($O(1)$ por candidato) y es factible por construcción.

  \textbf{Evidencia:} 30/30 soluciones iniciales cumplen las restricciones; $\approx 0.004$ ms por instancia.
\end{frame}

\begin{frame}{Greedy: historia del diseño}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \item \textbf{Ingenuo:} aporte $= f_T(t_{i-1}, t_i')$
        \begin{itemize}
          \item Ignora el efecto sobre la clase abierta
          \item Empuja los umbrales a la cola vacía del histograma
          \item $k = 2, 4, 6$ dan casi el mismo $f$
        \end{itemize}
        \item \textbf{Equiespaciar} la masa restante: casi no mejora
        \item \textbf{Sumar la masa restante} como una clase: $f_T(t_i', 256)$
      \end{enumerate}

\begin{frame}{Tabú Search: movimientos y diseño}
  \begin{itemize}
    \item Recibe la solución inicial $t = (t_1, \ldots, t_k)$ del Greedy
    \item \textbf{Movimiento:} sumar o restar 1 a un umbral $\Rightarrow$ a lo más $2k$ vecinos
    \begin{itemize}
      \item Se descartan los vecinos fuera de $[1, 254]$ o que repiten un umbral existente (factibles)
      \item Conserva el orden estricto y cambia solo dos clases
    \end{itemize}
    \item \textbf{Lista tabú:} cola FIFO de tuplas $(i, v)$, con $i$ el índice del umbral y $v$ el valor al que llegó
    \begin{itemize}
      \item Prohíbe volver a poner el umbral $i$ en el valor $v$
      \item Parte con los valores iniciales $(i, t_i)$ y guarda $k + T$ pares, con $T = 10$
    \end{itemize}
    \item \textbf{Selección:} el mejor vecino no tabú, aunque empeore la función objetivo
    \begin{itemize}
      \item \textbf{Aspiración:} un vecino tabú se acepta si supera al mejor valor observado
      \item Si todos son tabú, se elige el mejor de los tabú
    \end{itemize}
    \item \textbf{Término:} 1000 iteraciones; devuelve la mejor solución observada, no la última
  \end{itemize}
\end{frame}

\begin{frame}{Tabú Search: historia del diseño}
  \begin{columns}[T]
